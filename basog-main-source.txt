local SHOT_PACKET
do
    local bytes = {
        1, 235, 149, 185, 193,
        25, 207, 51, 194,
        16, 12, 158, 195,
        106, 161, 204, 193,
        100, 238, 52, 194,
        226, 106, 148, 195
    }
    local b = buffer.create(#bytes)
    for i = 1, #bytes do
        buffer.writeu8(b, i - 1, bytes[i])
    end
    SHOT_PACKET = b
end

local SCP_NAMES = {
    "SCP-049", "SCP-173", "SCP-096", "SCP-106",
    "SCP-457", "SCP-035", "SCP-966", "SCP-682",
}

local function getPlayerForSCP(scpName)
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player then
            local char = p.Character
            if char then
                if char.Name == scpName then return p end
                for _, obj in ipairs(char:GetChildren()) do
                    if obj.Name == scpName then return p end
                end
            end
        end
    end
    return nil
end

local function buildOptions()
    local opts = { "All", "Guard", "MTF" }
    for _, scpName in ipairs(SCP_NAMES) do
        local owner = getPlayerForSCP(scpName)
        if owner then
            table.insert(opts, scpName .. " (" .. owner.Name .. ")")
        else
            table.insert(opts, scpName .. " (empty)")
        end
    end
    return opts
end

local function playerMatchesTarget(targetPlayer)
    if AutoKillAll then return true end
    if targetPlayer.Team and AutoKillTargets[targetPlayer.Team.Name] then return true end
    local char = targetPlayer.Character
    if char then
        for _, scpName in ipairs(SCP_NAMES) do
            if AutoKillTargetSCPs[scpName] then
                if char.Name == scpName then return true end
                for _, obj in ipairs(char:GetChildren()) do
                    if obj.Name == scpName then return true end
                end
            end
        end
    end
    return false
end

--// SHOTS-PER-SHOT RESOLVER (weapon-based, premium-gated)
local function getEquippedTool()
    local char = player.Character
    if not char then return nil end
    return char:FindFirstChildOfClass("Tool")
end

local function getWeaponData()
    local tool = getEquippedTool()
    if not tool then return nil end
    local success, stats = pcall(function()
        if not WeaponStats then
            WeaponStats = require(ReplicatedStorage:WaitForChild('WeaponStats'))
        end
        return WeaponStats
    end)
    if not success or not stats then return nil end
    return stats[tool.Name], tool
end

local function getShotsForWeapon()
    if not HasPremiumKey then return 1 end
    local data = getWeaponData()
    if not data then return 1 end
    local shots = data.Shots or 1
    if data.Automatic and data.Shots == 1 then
        shots = data.MagSize or 1
    end
    return math.max(1, math.floor(shots))
end

-- How long to wait before firing the same target again.
-- Full-auto (whole-mag dump): wait the reload time.
-- Otherwise: wait the weapon's firerate.
local function getFireCooldown()
    local data = getWeaponData()
    if not data then return 0.1 end
    if not HasPremiumKey then return data.Firerate or 0.1 end
    if data.Automatic and data.Shots == 1 then
        return (data.ReloadTime or 2) + 0.1
    end
    return data.Firerate or 0.1
end

local function canFire()
    return tick() >= fireLock
end

local function markFired()
    fireLock = tick() + getFireCooldown()
end

local function fireAt(targetPlayer)
    if not canFire() then return end
    local character = targetPlayer.Character
    if not character then return end
    local rightArm = character:FindFirstChild("Right Arm")
    if not rightArm then return end

    local shots = getShotsForWeapon()
    for _ = 1, shots do
        GunShotEvent:FireServer("shoot", SHOT_PACKET, { targetPlayer }, { rightArm })
    end
    markFired()
end

--// METHOD 1 NO SPREAD (direct table edit, Spread = 0)
local function applyZeroSpread()
    local success = pcall(function()
        if not WeaponStats then
            WeaponStats = require(ReplicatedStorage:WaitForChild('WeaponStats'))
        end
        OriginalSpreads = {}
        for name, stats in pairs(WeaponStats) do
            if typeof(stats) == "table" and stats.Spread ~= nil then
                OriginalSpreads[name] = stats.Spread
                stats.Spread = 0
            end
        end
    end)
    if success then
        Rayfield:Notify({Title = "Zero Spread", Content = "Enabled", Duration = 2})
    else
        Rayfield:Notify({Title = "Error", Content = "Could not enable zero spread", Duration = 2})
    end
    return success
end

local function disableZeroSpread()
    if not WeaponStats then return end
    for name, original in pairs(OriginalSpreads) do
        if WeaponStats[name] then
            WeaponStats[name].Spread = original
        end
    end
    OriginalSpreads = {}
    Rayfield:Notify({Title = "Zero Spread", Content = "Disabled", Duration = 2})
end

--// Fly
local function startFly()
    if CFloop then CFloop:Disconnect() end
    local character = player.Character
    if not character then return end
    local humanoid = character:FindFirstChildOfClass('Humanoid')
    local head = character:WaitForChild("Head")
    if humanoid then humanoid.PlatformStand = true end
    head.Anchored = true

    CFloop = RunService.Heartbeat:Connect(function(deltaTime)
        if not player.Character then return end
        local hum = player.Character:FindFirstChildOfClass('Humanoid')
        local hd = player.Character:FindFirstChild("Head")
        if not hum or not hd then return end

        local moveDirection = hum.MoveDirection * (CFspeed * deltaTime)
        local headCFrame = hd.CFrame
        local cameraCFrame = Camera.CFrame
        local cameraOffset = headCFrame:ToObjectSpace(cameraCFrame).Position
        cameraCFrame = cameraCFrame * CFrame.new(-cameraOffset.X, -cameraOffset.Y, -cameraOffset.Z + 1)
        local cameraPosition = cameraCFrame.Position
        local headPosition = headCFrame.Position

        local objectSpaceVelocity = CFrame.new(cameraPosition, Vector3.new(headPosition.X, cameraPosition.Y, headPosition.Z)):VectorToObjectSpace(moveDirection)
        hd.CFrame = CFrame.new(headPosition) * (cameraCFrame - cameraPosition) * CFrame.new(objectSpaceVelocity)
    end)
    Rayfield:Notify({Title = "Invisible Fly", Content = "Enabled", Duration = 2})
end

local function stopFly()
    if CFloop then
        CFloop:Disconnect()
        CFloop = nil
    end
    local character = player.Character
    if not character then return end
    local humanoid = character:FindFirstChildOfClass('Humanoid')
    local head = character:FindFirstChild("Head")
    if humanoid then humanoid.PlatformStand = false end
    if head then head.Anchored = false end
    Rayfield:Notify({Title = "Invisible Fly", Content = "Disabled", Duration = 2})
end

--// HITBOX SYSTEM (hitbox-finished + MTF)
local function findTeam(keywordList)
    for _, team in ipairs(Teams:GetChildren()) do
        local name = team.Name:lower()
        for _, kw in ipairs(keywordList) do
            if name:find(kw, 1, true) then
                return team
            end
        end
    end
    return nil
end

local teamMap = {}
local dropdownOptions = {}

-- Build team map from keywords, ensuring MTF/Guard/SCP are always present
for label, keywords in pairs(TEAM_KEYWORDS) do
    local team = findTeam(keywords)
    if team then
        teamMap[label] = team
        table.insert(dropdownOptions, label)
    end
end

-- Fallback: if a keyword match failed (e.g. MTF team named differently), grab any remaining teams
if #dropdownOptions < 3 then
    for _, team in ipairs(Teams:GetChildren()) do
        local alreadyMapped = false
        for _, mapped in pairs(teamMap) do
            if mapped == team then alreadyMapped = true break end
        end
        if not alreadyMapped then
            teamMap[team.Name] = team
            table.insert(dropdownOptions, team.Name)
        end
    end
end

-- Guarantee MTF/Guard/SCP are in the dropdown even if no matching team object exists yet
local function ensureTeamOption(label)
    for _, opt in ipairs(dropdownOptions) do
        if opt == label then return end
    end
    table.insert(dropdownOptions, label)
end
ensureTeamOption("MTF")
ensureTeamOption("Guard")
ensureTeamOption("SCP")

local function resolveTeam(label)
    if teamMap[label] then return teamMap[label] end
    local keywords = TEAM_KEYWORDS[label]
    if keywords then
        local team = findTeam(keywords)
        if team then
            teamMap[label] = team
            return team
        end
    end
    for _, team in ipairs(Teams:GetChildren()) do
        if team.Name == label then
            teamMap[label] = team
            return team
        end
    end
    return nil
end

local function removeHitbox(character)
    if hitboxParts[character] then
        hitboxParts[character]:Destroy()
        hitboxParts[character] = nil
    end
    if hitboxDeathConns[character] then
        hitboxDeathConns[character]:Disconnect()
        hitboxDeathConns[character] = nil
    end
end

local function createHitbox(character)
    if not character then return end
    local hrp = character:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    removeHitbox(character)

    local part = Instance.new("Part")
    part.Name = "CustomHitbox"
    part.Size = HITBOX_SIZE
    part.Transparency = HITBOX_TRANSPARENCY
    part.Color = Color3.fromRGB(255, 0, 0)
    part.Material = Enum.Material.Neon
    part.CanCollide = false
    part.Massless = true
    part.Anchored = false
    part.CFrame = hrp.CFrame
    part.Parent = character

    local weld = Instance.new("WeldConstraint")
    weld.Part0 = part
    weld.Part1 = hrp
    weld.Parent = part

    hitboxParts[character] = part

    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if humanoid then
        hitboxDeathConns[character] = humanoid.Died:Connect(function()
            removeHitbox(character)
        end)
    end
end

local function shouldHaveHitbox(p)
    if not hitboxEnabled then return false end
    if not p.Team then return false end
    return activeTeams[p.Team.Name] == true
end

end

-- Physically-in-game check: real SpecialMesh, live part, part actually rendered.
local function get035MeshObject()
    local scp = get035Model()
    if not scp then return nil end
    if not scp.Parent then return nil end
    if not scp:IsDescendantOf(workspace) then return nil end

    for _, d in ipairs(scp:GetDescendants()) do
        if d:IsA("SpecialMesh") then
            local part = d.Parent
            if part and part:IsA("BasePart") and part:IsDescendantOf(workspace) then
                -- Part must actually be visible / rendered
                if part.Transparency < 1
                    and part.Position.Magnitude > 0
                    and d.MeshId ~= nil
                    and d.MeshId ~= "" then
                    return d
                end
            end
        end
    end
    return nil
end

local function get035TargetPart()
    local mesh = get035MeshObject()
    if not mesh then return nil end

--// INSTANT FAST FIRE (instant-firerate-done) - crosshair raycast + autoshoot
local function teamAllowed(targetPlayer)
    if not targetPlayer.Team then return false end
    return TargetTeams[targetPlayer.Team.Name] == true
end

local function playerFromHit(hit)
    if not hit then return nil end
    local model = hit:FindFirstAncestorOfClass("Model")
    while model do
        local p = Players:GetPlayerFromCharacter(model)
        if p then
            if p ~= player and teamAllowed(p) then
                return p
            end
            return nil
        end
        model = model:FindFirstAncestorOfClass("Model")
    end
    return nil
end

local function getTargetUnderCrosshair()
    if not Camera then return nil end
    local origin = Camera.CFrame.Position
    local direction = Camera.CFrame.LookVector * MaxRange

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = { player.Character }

    local result = Workspace:Raycast(origin, direction, params)
    local hitPlayer = playerFromHit(result and result.Instance)
    if hitPlayer then return hitPlayer end

    local offsets = {
        Vector3.new(AimTolerance, 0, 0),
        Vector3.new(-AimTolerance, 0, 0),
        Vector3.new(0, AimTolerance, 0),
        Vector3.new(0, -AimTolerance, 0),
    }
    for _, off in ipairs(offsets) do
        local fanOrigin = origin + Camera.CFrame:VectorToWorldSpace(off)
        local fanResult = Workspace:Raycast(fanOrigin, direction, params)
        local fanPlayer = playerFromHit(fanResult and fanResult.Instance)
        if fanPlayer then return fanPlayer end
    end
    return nil
end

local function fireBurstAt(targetPlayer)
    if not GunShotEvent or not targetPlayer then return end

    local targetChar = targetPlayer.Character

    local shots = getShotsForWeapon()
    for _ = 1, shots do
        GunShotEvent:FireServer("shoot", SHOT_PACKET, { targetPlayer }, { rightArm })
    end
    markFired()
end

    local lastKey = ""
    while true do
        task.wait(3)
        local opts = buildOptions()
        local key = table.concat(opts, "|")
        if key ~= lastKey then
            lastKey = key
            if AutoKillDropdown then
                AutoKillDropdown:Refresh(opts)
            end
        end
    end
end)

local function findExitPart()
    local pocketDimension = Workspace:FindFirstChild("Pocket Dimension")
    if not pocketDimension then return nil end
    local model = pocketDimension:FindFirstChild("Model")
    if not model then return nil end
    return model:FindFirstChild("Exit")
end

local function createExitESP()
    if ExitESPObjects.Highlight then ExitESPObjects.Highlight:Destroy() end
    if ExitESPObjects.Billboard then ExitESPObjects.Billboard:Destroy() end
    if ExitESPObjects.Connection then ExitESPObjects.Connection:Disconnect() end

    local targetPart = findExitPart()
    if not targetPart then return false end

    ExitESPObjects.Highlight = Instance.new("Highlight")
    ExitESPObjects.Highlight.Name = "ExitESP_Highlight"
    ExitESPObjects.Highlight.FillColor = Color3.fromRGB(0, 255, 0)
    ExitESPObjects.Highlight.OutlineColor = Color3.fromRGB(0, 200, 0)
    ExitESPObjects.Highlight.FillTransparency = 0.3
    ExitESPObjects.Highlight.OutlineTransparency = 0
    ExitESPObjects.Highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    ExitESPObjects.Highlight.Adornee = targetPart
    ExitESPObjects.Highlight.Parent = CoreGui

    ExitESPObjects.Billboard = Instance.new("BillboardGui")
    ExitESPObjects.Billboard.Name = "ExitESP_Label"
    ExitESPObjects.Billboard.Size = UDim2.new(0, 300, 0, 60)
    ExitESPObjects.Billboard.StudsOffset = Vector3.new(0, 5, 0)
    ExitESPObjects.Billboard.AlwaysOnTop = true
    ExitESPObjects.Billboard.Adornee = targetPart

    local label = Instance.new("TextLabel")
    label.Name = "ExitText"
    label.Size = UDim2.new(1, 0, 0.6, 0)
    label.BackgroundTransparency = 1
    label.Text = "EXIT"
    label.TextColor3 = Color3.fromRGB(0, 255, 0)
    label.TextStrokeTransparency = 0
    label.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
    label.Font = Enum.Font.GothamBold
    label.TextSize = 28
    label.Parent = ExitESPObjects.Billboard

    local distanceLabel = Instance.new("TextLabel")
    distanceLabel.Name = "DistanceText"
    distanceLabel.Size = UDim2.new(1, 0, 0.4, 0)
    distanceLabel.Position = UDim2.new(0, 0, 0.6, 0)
    distanceLabel.BackgroundTransparency = 1
    distanceLabel.Text = "Distance: calculating..."
    distanceLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
    distanceLabel.TextStrokeTransparency = 0
    distanceLabel.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
    distanceLabel.Font = Enum.Font.GothamBold
    distanceLabel.TextSize = 18
    distanceLabel.Parent = ExitESPObjects.Billboard

    ExitESPObjects.Billboard.Parent = CoreGui

    ExitESPObjects.Connection = RunService.RenderStepped:Connect(function()
        if not targetPart or not targetPart.Parent then
            local newPart = findExitPart()
            if newPart then
                targetPart = newPart
                ExitESPObjects.Highlight.Adornee = targetPart
                ExitESPObjects.Billboard.Adornee = targetPart
            end
            return
        end
        if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            local distance = (player.Character.HumanoidRootPart.Position - targetPart.Position).Magnitude
            distanceLabel.Text = string.format("Distance: %.1f studs", distance)
        end
    end)
    return true
end

local function removeExitESP()
    if ExitESPObjects.Highlight then ExitESPObjects.Highlight:Destroy() end
    if ExitESPObjects.Billboard then ExitESPObjects.Billboard:Destroy() end
    if ExitESPObjects.Connection then ExitESPObjects.Connection:Disconnect() end
    ExitESPObjects = {}
end

local TeamColors = {
    ["Guard"] = Color3.fromRGB(0, 0, 255),
    ["MTF"] = Color3.fromRGB(0, 255, 0),
    ["SCP"] = Color3.fromRGB(255, 0, 0),
}
local DefaultColor = Color3.fromRGB(255, 255, 255)

local function applyTeamChams(character)
    if not character then return end
    for _, v in ipairs(character:GetChildren()) do
        if v:IsA("Highlight") and v.Name == "TeamChams" then v:Destroy() end
    end
    local targetPlayer = Players:GetPlayerFromCharacter(character)
    if not targetPlayer then return end
    local teamName = targetPlayer.Team and targetPlayer.Team.Name or "None"
    if not SelectedESPTeams[teamName] then return end
    local color = TeamColors[teamName] or DefaultColor

    local highlight = Instance.new("Highlight")
    highlight.Name = "TeamChams"
    highlight.FillColor = color
    highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
    highlight.FillTransparency = 0.5
    highlight.OutlineTransparency = 0
    highlight.Parent = character
end

local function removeTeamChams(character)
    if not character then return end
    for _, v in ipairs(character:GetChildren()) do
        if v:IsA("Highlight") and v.Name == "TeamChams" then v:Destroy() end
    end
end

local function refreshAllChams()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player and p.Character then
            removeTeamChams(p.Character)
            applyTeamChams(p.Character)
        end
    end
end

local function startTeamESP()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player then
            if p.Character then applyTeamChams(p.Character) end
            local charConn = p.CharacterAdded:Connect(applyTeamChams)
            local teamConn = p:GetPropertyChangedSignal("Team"):Connect(function()
                if p.Character then
                    removeTeamChams(p.Character)
                    applyTeamChams(p.Character)
                end
            end)
            TeamESPConnections[p] = {charConn, teamConn}
        end
    end
    TeamESPConnections.PlayerAdded = Players.PlayerAdded:Connect(function(p)
        if p == player then return end
        local charConn = p.CharacterAdded:Connect(applyTeamChams)
        local teamConn = p:GetPropertyChangedSignal("Team"):Connect(function()
            if p.Character then
                removeTeamChams(p.Character)
                applyTeamChams(p.Character)
            end
        end)
        TeamESPConnections[p] = {charConn, teamConn}
    end)
end

local function stopTeamESP()
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Character then removeTeamChams(p.Character) end
    end
    for p, conns in pairs(TeamESPConnections) do
        if p ~= "PlayerAdded" then
            for _, conn in ipairs(conns) do conn:Disconnect() end
        end
    end
    if TeamESPConnections.PlayerAdded then TeamESPConnections.PlayerAdded:Disconnect() end
    TeamESPConnections = {}
end
